```components/ui/Button.tsx
import React from 'react'
type ButtonProps = {
  children: React.ReactNode
  variant?: 'primary' | 'secondary' | 'outline'
  size?: 'sm' | 'md' | 'lg'
  className?: string
  onClick?: () => void
}
const Button: React.FC<ButtonProps> = ({
  children,
  variant = 'primary',
  size = 'md',
  className = '',
  onClick,
  ...props
}) => {
  const baseStyles =
    'inline-flex items-center justify-center font-medium rounded-md transition-all duration-200 relative overflow-hidden'
  const variantStyles = {
    primary:
      'bg-red-600 hover:bg-red-700 text-white shadow-lg shadow-red-600/20 hover:shadow-red-600/40 border border-red-500',
    secondary:
      'bg-gray-800 hover:bg-gray-700 text-white shadow-lg shadow-gray-900/30 border border-gray-700',
    outline:
      'bg-transparent hover:bg-gray-800/50 text-white border border-gray-600 hover:border-red-500',
  }
  const sizeStyles = {
    sm: 'text-sm py-1 px-3',
    md: 'text-base py-2 px-5',
    lg: 'text-lg py-3 px-7',
  }
  const glowEffect =
    variant === 'primary'
      ? 'after:absolute after:content-[""] after:w-full after:h-full after:bg-gradient-to-r after:from-transparent after:via-white/10 after:to-transparent after:top-0 after:left-[-100%] after:animate-shine after:z-10'
      : ''
  return (
    <button
      className={`${baseStyles} ${variantStyles[variant]} ${sizeStyles[size]} ${glowEffect} ${className}`}
      onClick={onClick}
      {...props}
    >
      {children}
    </button>
  )
}
export default Button
```
